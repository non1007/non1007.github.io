// --------------- POST EFFECT DEFINITION --------------- //
/**
 * @class
 * @name VignetteEffect
 * @classdesc Implements a vignette post processing effect.
 * [EN] Post effect that darkens the image towards the edges, simulating lens vignetting.  
 * [JP] 画像の端を暗くするビネット効果を適用し、レンズのビネット現象を再現するポストエフェクト。
 *
 * @augments PostEffect
 * @param {GraphicsDevice} graphicsDevice
 * [EN] The graphics device of the application.  
 * [JP] アプリケーションの GraphicsDevice。
 *
 * @property {number} offset
 * [EN] Controls how far from the center the vignette effect starts.  
 * [JP] ビネット効果が画面中心からどの程度離れた位置で始まるかを制御する。  
 *
 * @property {number} darkness
 * [EN] Intensity of the vignette darkness at the edges.  
 * [JP] 画面端に適用される暗さの強さ。
 */
class VignetteEffect extends pc.PostEffect {
    constructor(graphicsDevice) {
        super(graphicsDevice); // ← 必須
        // Shader
        const luminosityFrag = `
            uniform sampler2D uColorBuffer;
            uniform float uDarkness;
            uniform float uOffset;
            varying vec2 vUv0;
            void main() {
                vec4 texel = texture2D(uColorBuffer, vUv0);
                vec2 uv = (vUv0 - vec2(0.5)) * vec2(uOffset);
                gl_FragColor = vec4(mix(texel.rgb, vec3(1.0 - uDarkness), dot(uv, uv)), texel.a);
            }
        `;

        this.vignetteShader = pc.ShaderUtils.createShader(graphicsDevice, {
            uniqueName: 'VignetteShader',
            attributes: { aPosition: pc.SEMANTIC_POSITION },
            vertexGLSL: pc.PostEffect.quadVertexShader,
            fragmentGLSL: luminosityFrag
        });

        this.offset = 1;
        this.darkness = 1;
    }

    render(inputTarget, outputTarget, rect) {
        const device = this.device;
        const scope = device.scope;

        scope.resolve('uColorBuffer').setValue(inputTarget.colorBuffer);
        scope.resolve('uOffset').setValue(this.offset);
        scope.resolve('uDarkness').setValue(this.darkness);
        this.drawQuad(outputTarget, this.vignetteShader, rect);
    }
}

// ----------------- SCRIPT DEFINITION ------------------ //
/**
 * @class
 * @name Vignette
 * @classdesc Script component wrapper for {@link VignetteEffect}.
 * [EN] Script to use VignetteEffect inside the PlayCanvas Editor.  
 * [JP] PlayCanvas エディタで VignetteEffect を利用するためのスクリプトコンポーネント。
 *
 * @property {number} offset
 * [EN] Controls how far from the center the vignette starts (0 = center, higher = further).  
 * [JP] ビネット効果が画面中心から始まる位置を制御（0 = 中央、値が大きいほど外側から）。  
 *
 * @property {number} darkness
 * [EN] Darkness intensity at the screen edges.  
 * [JP] 画面端に適用される暗さの強さ。
 */
var Vignette = pc.createScript('vignette');

Vignette.attributes.add('offset', {
    type: 'number',
    default: 1,
    min: 0,
    title: 'Offset / オフセット',
    description: 'EN: Distance from center where vignette starts. / JP: ビネット効果が中心から始まる距離。'
});

Vignette.attributes.add('darkness', {
    type: 'number',
    default: 1,
    title: 'Darkness / 暗さ',
    description: 'EN: Intensity of vignette darkness at the edges. / JP: 画面端に適用される暗さの強さ。'
});

// initialize code called once per entity
Vignette.prototype.initialize = function () {
    this.effect = new VignetteEffect(this.app.graphicsDevice);
    this.effect.offset = this.offset;
    this.effect.darkness = this.darkness;

    this.on('attr', function (name, value) {
        this.effect[name] = value;
    }, this);

    const queue = this.entity.camera.postEffects;
    queue.addEffect(this.effect);

    this.on('state', function (enabled) {
        if (enabled) {
            queue.addEffect(this.effect);
        } else {
            queue.removeEffect(this.effect);
        }
    });

    this.on('destroy', function () {
        queue.removeEffect(this.effect);
    });
};
