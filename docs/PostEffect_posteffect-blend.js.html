// --------------- POST EFFECT DEFINITION --------------- //
/**
 * @class
 * @name BlendEffect
 * @classdesc
 * [EN] Post effect that blends the input render target with another texture.  
 * [JP] 入力レンダーターゲットを別のテクスチャとブレンドするポストエフェクト。
 *
 * @description
 * [EN] Creates a new instance of the blend post effect.  
 * [JP] Blend ポストエフェクトの新しいインスタンスを作成します。
 *
 * @augments pc.PostEffect
 *
 * @param {pc.GraphicsDevice} graphicsDevice -  
 * [EN] The graphics device of the application.  
 * [JP] アプリケーションの GraphicsDevice。
 *
 * @property {pc.Texture} blendMap  
 * [EN] Texture used to blend with the input render target.  
 * [JP] 入力レンダーターゲットとブレンドするテクスチャ。
 *
 * @property {number} mixRatio  
 * [EN] Blend factor between input and blendMap (0.0 = only input, 1.0 = only blendMap).  
 * [JP] 入力と blendMap のブレンド係数（0.0 = 入力のみ、1.0 = blendMap のみ）。
 */
// --------------- POST EFFECT DEFINITION --------------- //
class BlendEffect extends pc.PostEffect {
    /**
     * @param {pc.GraphicsDevice} graphicsDevice
     */
    constructor(graphicsDevice) {
        super(graphicsDevice);

        const fshader = `
            uniform float uMixRatio;
            uniform sampler2D uColorBuffer;
            uniform sampler2D uBlendMap;
            varying vec2 vUv0;

            void main(void) {
                vec4 texel1 = texture2D(uColorBuffer, vUv0);
                vec4 texel2 = texture2D(uBlendMap, vUv0);
                gl_FragColor = mix(texel1, texel2, uMixRatio);
            }
        `;

        this.shader = pc.ShaderUtils.createShader(graphicsDevice, {
            uniqueName: 'BlendShader',
            attributes: { aPosition: pc.SEMANTIC_POSITION },
            vertexGLSL: pc.PostEffect.quadVertexShader,
            fragmentGLSL: fshader
        });

        // uniforms
        this.mixRatio = 0.5;
        this.blendMap = new pc.Texture(graphicsDevice);
        this.blendMap.name = 'pe-blend';
    }

    render(inputTarget, outputTarget, rect) {
        const device = this.device;
        const scope = device.scope;

        scope.resolve('uMixRatio').setValue(this.mixRatio);
        scope.resolve('uColorBuffer').setValue(inputTarget.colorBuffer);
        scope.resolve('uBlendMap').setValue(this.blendMap);

        this.drawQuad(outputTarget, this.shader, rect);
    }
}

// ----------------- SCRIPT DEFINITION ------------------ //
/**
 * @class
 * @name blend
 * @classdesc
 * [EN] PlayCanvas script that applies `BlendEffect` to a camera.  
 * [JP] カメラに `BlendEffect` を適用する PlayCanvas スクリプト。
 *
 * @property {number} mixRatio - Mix Ratio / ブレンド比率  
 * [EN] Blend factor between input and blendMap. (0 = only input, 1 = only blendMap)  
 * [JP] 入力と blendMap のブレンド係数。（0 = 入力のみ、1 = blendMap のみ）
 *
 * @property {pc.Asset} blendMap - Blend Map / ブレンドマップ  
 * [EN] Texture asset used for blending.  
 * [JP] ブレンドに使用するテクスチャアセット。
 */
var Blend = pc.createScript('blend');

Blend.attributes.add('mixRatio', {
    type: 'number',
    default: 0.5,
    min: 0,
    max: 1,
    title: 'Mix Ratio / ブレンド比率',
    description: 'EN: Blend factor between input and blendMap. / JP: 入力と blendMap のブレンド係数。'
});

Blend.attributes.add('blendMap', {
    type: 'asset',
    assetType: 'texture',
    title: 'Blend Map / ブレンドマップ',
    description: 'EN: Texture asset used for blending. / JP: ブレンドに使用するテクスチャアセット。'
});

Blend.prototype.initialize = function () {
    this.effect = new BlendEffect(this.app.graphicsDevice);
    this.effect.mixRatio = this.mixRatio;
    if (this.blendMap) {
        this.effect.blendMap = this.blendMap.resource;
    }

    var queue = this.entity.camera.postEffects;
    queue.addEffect(this.effect);

    this.on('state', function (enabled) {
        if (enabled) {
            queue.addEffect(this.effect);
        } else {
            queue.removeEffect(this.effect);
        }
    });

    this.on('destroy', function () {
        queue.removeEffect(this.effect);
    });

    this.on('attr:mixRatio', function (value) {
        this.effect.mixRatio = value;
    }, this);

    this.on('attr:blendMap', function (value) {
        this.effect.blendMap = value ? value.resource : null;
    }, this);
};
