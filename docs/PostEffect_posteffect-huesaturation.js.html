// --------------- POST EFFECT DEFINITION --------------- //
/**
 * @class
 * @name HueSaturationEffect
 * @classdesc Allows hue and saturation adjustment of the input render target.
 * [EN] Adjusts the hue and saturation of the rendered image.  
 * [JP] レンダリングされた画像の色相と彩度を調整します。
 *
 * @augments PostEffect
 * @param {GraphicsDevice} graphicsDevice  
 * [EN] The graphics device of the application.  
 * [JP] アプリケーションの GraphicsDevice。
 *
 * @property {number} hue  
 * [EN] Hue adjustment factor. Range: -1 to 1 (negative = rotate left, positive = rotate right).  
 * [JP] 色相の調整係数。範囲: -1 ～ 1（負の値 = 左回転, 正の値 = 右回転）。
 *
 * @property {number} saturation  
 * [EN] Saturation adjustment factor. Range: -1 (grayscale) to 1 (extra vivid).  
 * [JP] 彩度の調整係数。範囲: -1（白黒）～ 1（鮮やか）。
 */
class HueSaturationEffect extends pc.PostEffect {
    constructor(graphicsDevice) {
        super(graphicsDevice);

        const fshader = `
            uniform sampler2D uColorBuffer;
            uniform float uHue;
            uniform float uSaturation;
            varying vec2 vUv0;

            void main() {
                vec3 color = texture2D(uColorBuffer, vUv0).rgb;

                // Hue adjustment
                float angle = uHue * 3.14159265;
                float s = sin(angle), c = cos(angle);
                vec3 weights = (vec3(2.0 * c, -sqrt(3.0) * s - c, sqrt(3.0) * s - c) + 1.0) / 3.0;
                color = vec3(
                    dot(color, weights.xyz),
                    dot(color, weights.zxy),
                    dot(color, weights.yzx)
                );

                // Saturation adjustment
                float average = (color.r + color.g + color.b) / 3.0;
                if (uSaturation > 0.0) {
                    color += (average - color) * (1.0 - 1.0 / (1.001 - uSaturation));
                } else {
                    color += (average - color) * (-uSaturation);
                }

                gl_FragColor = vec4(color, 1.0);
            }
        `;

        this.shader = pc.ShaderUtils.createShader(graphicsDevice, {
            uniqueName: 'HueSaturationShader',
            attributes: { aPosition: pc.SEMANTIC_POSITION },
            vertexGLSL: pc.PostEffect.quadVertexShader,
            fragmentGLSL: fshader
        });

        // uniforms
        this.hue = 0;
        this.saturation = 0;
    }

    render(inputTarget, outputTarget, rect) {
        const scope = this.device.scope;
        scope.resolve('uHue').setValue(this.hue);
        scope.resolve('uSaturation').setValue(this.saturation);
        scope.resolve('uColorBuffer').setValue(inputTarget.colorBuffer);
        this.drawQuad(outputTarget, this.shader, rect);
    }
}

/**
 * @class
 * @name HueSaturation
 * @classdesc PlayCanvas script wrapper for {@link HueSaturationEffect}.
 * [EN] Script component to use HueSaturationEffect in the PlayCanvas Editor.  
 * [JP] PlayCanvas エディタで HueSaturationEffect を利用するためのスクリプトコンポーネント。
 *
 * @property {number} hue  
 * [EN] Hue adjustment factor (-1 to 1).  
 * [JP] 色相の調整係数（-1 ～ 1）。
 *
 * @property {number} saturation  
 * [EN] Saturation adjustment factor (-1 to 1).  
 * [JP] 彩度の調整係数（-1 ～ 1）。
 */
var HueSaturation = pc.createScript('hueSaturation');

HueSaturation.attributes.add('hue', {
    type: 'number',
    default: 0,
    min: -1,
    max: 1,
    title: 'Hue / 色相',
    description: 'EN: Hue adjustment factor. Range: -1 to 1. / JP: 色相の調整係数。範囲: -1 ～ 1。'
});

HueSaturation.attributes.add('saturation', {
    type: 'number',
    default: 0,
    min: -1,
    max: 1,
    title: 'Saturation / 彩度',
    description: 'EN: Saturation adjustment factor. Range: -1 (grayscale) to 1 (extra vivid). / JP: 彩度の調整係数。範囲: -1（白黒）～ 1（鮮やか）。'
});

HueSaturation.prototype.initialize = function () {
    this.effect = new HueSaturationEffect(this.app.graphicsDevice);
    this.effect.hue = this.hue;
    this.effect.saturation = this.saturation;

    this.on('attr', function (name, value) {
        this.effect[name] = value;
    }, this);

    const queue = this.entity.camera.postEffects;
    queue.addEffect(this.effect);

    this.on('state', function (enabled) {
        if (enabled) {
            queue.addEffect(this.effect);
        } else {
            queue.removeEffect(this.effect);
        }
    });

    this.on('destroy', function () {
        queue.removeEffect(this.effect);
    });
};
