/**
 * @class
 * @name BrightnessContrastEffect
 * @classdesc
 * [EN] Post effect that changes the brightness and contrast of the input render target.  
 * [JP] 入力レンダーターゲットの明るさとコントラストを変更するポストエフェクト。
 *
 * @description
 * [EN] Creates a new instance of the BrightnessContrastEffect post effect.  
 * [JP] BrightnessContrastEffect ポストエフェクトの新しいインスタンスを作成します。
 *
 * @augments PostEffect
 * @param {GraphicsDevice} graphicsDevice
 * [EN] The graphics device of the application.  
 * [JP] アプリケーションの GraphicsDevice。
 *
 * @property {number} brightness  
 * [EN] Controls the brightness. Range: -1 to 1. (-1 = solid black, 0 = no change, 1 = solid white)  
 * [JP] 明るさを制御します。範囲は -1 から 1 です。(-1 = 完全な黒, 0 = 変化なし, 1 = 完全な白)
 *
 * @property {number} contrast  
 * [EN] Controls the contrast. Range: -1 to 1. (-1 = solid gray, 0 = no change, 1 = maximum contrast)  
 * [JP] コントラストを制御します。範囲は -1 から 1 です。(-1 = 完全な灰色, 0 = 変化なし, 1 = 最大コントラスト)
 */
class BrightnessContrastEffect extends pc.PostEffect {
    constructor(graphicsDevice) {
        super(graphicsDevice);

        // Shader
        const fshader = `
            uniform sampler2D uColorBuffer;
            uniform float uBrightness;
            uniform float uContrast;
            
            varying vec2 vUv0;
            
            void main() {
                gl_FragColor = texture2D(uColorBuffer, vUv0);
                gl_FragColor.rgb += uBrightness;

                if (uContrast > 0.0) {
                    gl_FragColor.rgb = (gl_FragColor.rgb - 0.5) / (1.0 - uContrast) + 0.5;
                } else {
                    gl_FragColor.rgb = (gl_FragColor.rgb - 0.5) * (1.0 + uContrast) + 0.5;
                }
            }
        `;

        this.shader = pc.ShaderUtils.createShader(graphicsDevice, {
            uniqueName: 'BrightnessContrastShader',
            attributes: { aPosition: pc.SEMANTIC_POSITION },
            vertexGLSL: pc.PostEffect.quadVertexShader,
            fragmentGLSL: fshader
        });

        // Uniforms
        this.brightness = 0;
        this.contrast = 0;
    }

    render(inputTarget, outputTarget, rect) {
        const device = this.device;
        const scope = device.scope;

        scope.resolve('uBrightness').setValue(this.brightness);
        scope.resolve('uContrast').setValue(this.contrast);
        scope.resolve('uColorBuffer').setValue(inputTarget.colorBuffer);

        this.drawQuad(outputTarget, this.shader, rect);
    }
}

 *      範囲: -1（真っ黒）、0（変化なし）、1（真っ白）。
 *
 * @property {number} contrast
 * [EN] Controls the contrast of the render target.

BrightnessContrast.attributes.add('brightness', {
    type: 'number',
    default: 0,
    min: -1,
    max: 1,
    title: 'Brightness / 明るさ',

BrightnessContrast.attributes.add('contrast', {
    type: 'number',
    default: 0,
    min: -1,
    max: 1,
    title: 'Contrast / コントラスト',
    description: 'EN: Contrast adjustment (-1 = gray, 0 = no change, 1 = max contrast). / JP: コントラストの調整（-1 = 灰色, 0 = 変化なし, 1 = 最大コントラスト）。'
});

BrightnessContrast.prototype.initialize = function () {
    this.effect = new BrightnessContrastEffect(this.app.graphicsDevice);
    this.effect.brightness = this.brightness;
    this.effect.contrast = this.contrast;

    this.on('attr', function (name, value) {
        this.effect[name] = value;
    }, this);

    const queue = this.entity.camera.postEffects;
    queue.addEffect(this.effect);

    this.on('state', function (enabled) {
        if (enabled) {
            queue.addEffect(this.effect);
        } else {
            queue.removeEffect(this.effect);
        }
    });

    this.on('destroy', function () {
        queue.removeEffect(this.effect);
    });
};
