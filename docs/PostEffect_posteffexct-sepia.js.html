/**
 * @class
 * @name SepiaEffect
 * @classdesc
 * [EN] Applies a sepia color filter post effect.  
 * [JP] セピア色のカラーフィルタポストエフェクトを適用するポストエフェクト。
 *
 * @description
 * [EN] This effect tints the rendered scene with a sepia tone, often used to give a nostalgic or cinematic look.  
 * [JP] レンダリングされたシーンをセピア調に着色します。ノスタルジックや映画的な雰囲気を演出するために使用されます。
 *
 * Example / 使用例:
 * ```javascript
 * var effect = new SepiaEffect(app.graphicsDevice);
 * effect.amount = 0.8;
 * cameraEntity.camera.postEffects.addEffect(effect);
 * ```
 *
 * @augments pc.PostEffect
 *
 * @param {pc.GraphicsDevice} graphicsDevice  
 * [EN] The graphics device of the application.  
 * [JP] アプリケーションの GraphicsDevice。
 *
 * @property {number} amount  
 * [EN] Intensity of the sepia effect (range 0.0–1.0).  
 * [JP] セピア効果の強さ（範囲 0.0〜1.0）。
 */
class SepiaEffect extends pc.PostEffect {
    constructor(graphicsDevice) {
        super(graphicsDevice); // ← 必須

        const fshader = `
            uniform float uAmount;
            uniform sampler2D uColorBuffer;
            varying vec2 vUv0;
            void main() {
                vec4 color = texture2D(uColorBuffer, vUv0);
                vec3 c = color.rgb;
                color.r = dot(c, vec3(1.0 - 0.607 * uAmount, 0.769 * uAmount, 0.189 * uAmount));
                color.g = dot(c, vec3(0.349 * uAmount, 1.0 - 0.314 * uAmount, 0.168 * uAmount));
                color.b = dot(c, vec3(0.272 * uAmount, 0.534 * uAmount, 1.0 - 0.869 * uAmount));
                gl_FragColor = vec4(min(vec3(1.0), color.rgb), color.a);
            }
        `;

        this.shader = pc.ShaderUtils.createShader(graphicsDevice, {
            uniqueName: 'SepiaShader',
            attributes: { aPosition: pc.SEMANTIC_POSITION },
            vertexGLSL: pc.PostEffect.quadVertexShader,
            fragmentGLSL: fshader
        });

        // Uniform
        this.amount = 1;
    }

    render(inputTarget, outputTarget, rect) {
        const scope = this.device.scope;
        scope.resolve('uAmount').setValue(this.amount);
        scope.resolve('uColorBuffer').setValue(inputTarget.colorBuffer);
        this.drawQuad(outputTarget, this.shader, rect);
    }
}

// ----------------- SCRIPT DEFINITION ------------------ //
/**
 * @class
 * @name sepia
 * @classdesc
 * [EN] PlayCanvas script that attaches `SepiaEffect` to a camera.  
 * [JP] カメラに `SepiaEffect` を追加する PlayCanvas スクリプト。
 *
 * @description
 * [EN] Attach this script to a camera entity to apply the sepia post effect.  
 * Use the `amount` attribute to control the intensity of the effect.  
 *
 * [JP] このスクリプトをカメラエンティティにアタッチすると、セピアポストエフェクトが有効になります。  
 * `amount` 属性を使って効果の強さを調整できます。
 *
 * @property {number} amount  
 * [EN] Intensity of the sepia effect (0.0 = none, 1.0 = full).  
 * [JP] セピア効果の強さ（0.0 = 無効, 1.0 = 最大）。
 */
var Sepia = pc.createScript('sepia');

Sepia.attributes.add('amount', {
    type: 'number',
    default: 1,
    min: 0,
    max: 1,
    title: 'Amount / 強さ',
    description: 'EN: Intensity of the sepia effect. / JP: セピア効果の強さ。'
});

// initialize code called once per entity
Sepia.prototype.initialize = function () {
    this.effect = new SepiaEffect(this.app.graphicsDevice);
    this.effect.amount = this.amount;

    this.on('attr:amount', function (value) {
        this.effect.amount = value;
    }, this);

    const queue = this.entity.camera.postEffects;
    queue.addEffect(this.effect);

    this.on('state', function (enabled) {
        if (enabled) {
            queue.addEffect(this.effect);
        } else {
            queue.removeEffect(this.effect);
        }
    });

    this.on('destroy', function () {
        queue.removeEffect(this.effect);
    });
};
