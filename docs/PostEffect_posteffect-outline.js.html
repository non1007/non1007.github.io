/**
 * @class
 * @name OutlinePostEffect
 * @classdesc
 * [EN] Post effect that draws outlines around objects using depth buffer.  
 * [JP] 深度バッファを用いてオブジェクトにアウトラインを描画するポストエフェクト。
 *
 * @description
 * [EN] This effect detects edges by comparing depth differences of neighboring pixels.  
 * Useful for highlighting 3D models or gameplay elements.  
 *
 * [JP] 隣接するピクセルの深度差を比較して輪郭を検出し、アウトラインを描画します。  
 * キャラクターやアイテムの強調表示などに利用できます。
 *
 * Usage / 使用例:
 * ```javascript
 * var effect = new OutlinePostEffect(app.graphicsDevice);
 * effect.outlineThickness = 2.0;
 * effect.outlineColor = [1, 0, 0]; // 赤いアウトライン
 * cameraEntity.camera.postEffects.addEffect(effect);
 * ```
 *
 * @augments PostEffect
 *
 * @param {GraphicsDevice} device
 * [EN] The graphics device, usually `app.graphicsDevice`.  
 * [JP] 利用中の GraphicsDevice。通常は `app.graphicsDevice` を渡します。
 *
 * @property {number} outlineThickness
 * [EN] Outline thickness in pixels.  
 * [JP] アウトラインの太さ（ピクセル単位）。
 *
 * @property {number[]} outlineColor
 * [EN] Outline color in RGB (array of 3 floats).  
 * [JP] アウトラインの描画色（RGB配列、0〜1の値）。
 *
 * @property {number} outlineDepthThreshold
 * [EN] Minimum depth difference to trigger outline detection.  
 * [JP] 輪郭を検出するための深度差のしきい値。
 *
 * @property {number} nearClip
 * [EN] Near clipping plane of the camera.  
 * [JP] カメラのニアクリップ距離。
 *
 * @property {number} farClip
 * [EN] Far clipping plane of the camera.  
 * [JP] カメラのファークリップ距離。
 *
 * @property {number} maxOutlineDistance
 * [EN] Maximum distance where outline is visible (fades beyond this).  
 * [JP] アウトラインが表示される最大距離（これを超えるとフェードアウト）。
 */
class OutlinePostEffect extends pc.PostEffect {
    constructor(device) {
        super(device);

        this.shader = pc.ShaderUtils.createShader(device, {
            uniqueName: 'outline-post-effect',
            attributes: {
                aPosition: pc.SEMANTIC_POSITION
            },
            vertexGLSL: [
                "attribute vec2 aPosition;",
                "varying vec2 vUv0;",
                "void main(void) {",
                "    gl_Position = vec4(aPosition, 0.0, 1.0);",
                "    vUv0 = (aPosition.xy + 1.0) * 0.5;",
                "}"
            ].join("\n"),
            fragmentGLSL: [
                "precision " + device.precision + " float;",
                "uniform sampler2D uColorBuffer;",
                "uniform sampler2D uDepthBuffer;",
                "uniform vec2 uResolution;",
                "uniform float uOutlineThickness;",
                "uniform vec3 uOutlineColor;",
                "uniform float uOutlineDepthThreshold;",
                "uniform float uNearClip;",
                "uniform float uFarClip;",
                "uniform float uMaxOutlineDistance;",
                "varying vec2 vUv0;",

                "float linearizeDepth(float depth) {",
                "    float z = depth * 2.0 - 1.0;",
                "    return (2.0 * uNearClip * uFarClip) / (uFarClip + uNearClip - z * (uFarClip - uNearClip));",
                "}",

                "float getSceneDepth(vec2 uv) {",
                "    float d = texture2D(uDepthBuffer, uv).r;",
                "    return d;",
                "}",

                "void main() {",
                "    vec2 texel = vec2(1.0) / uResolution;",
                "    float centerD = getSceneDepth(vUv0);",
                "    if (centerD >= 0.99999) {",
                "       gl_FragColor = texture2D(uColorBuffer, vUv0);",
                "       return;",
                "    }",
                "    float leftD  = getSceneDepth(vUv0 + vec2(-texel.x * uOutlineThickness, 0.0));",
                "    float rightD = getSceneDepth(vUv0 + vec2(texel.x * uOutlineThickness, 0.0));",
                "    float upD    = getSceneDepth(vUv0 + vec2(0.0, texel.y * uOutlineThickness));",
                "    float downD  = getSceneDepth(vUv0 + vec2(0.0, -texel.y * uOutlineThickness));",

                "    if (leftD  >= 0.99999) leftD  = centerD;",
                "    if (rightD >= 0.99999) rightD = centerD;",
                "    if (upD    >= 0.99999) upD    = centerD;",
                "    if (downD  >= 0.99999) downD  = centerD;",

                "    float depthDiff = 0.0;",
                "    depthDiff += step(uOutlineDepthThreshold, abs(centerD - leftD));",
                "    depthDiff += step(uOutlineDepthThreshold, abs(centerD - rightD));",
                "    depthDiff += step(uOutlineDepthThreshold, abs(centerD - upD));",
                "    depthDiff += step(uOutlineDepthThreshold, abs(centerD - downD));",

                "    float linearCenter = linearizeDepth(centerD);",
                "    float fade = clamp(1.0 - (linearCenter / uMaxOutlineDistance), 0.0, 1.0);",
                "    float edge = depthDiff * fade;",
                "    vec4 sceneColor = texture2D(uColorBuffer, vUv0);",

                "    if (edge > 0.0) {",
                "        gl_FragColor = vec4(uOutlineColor, 1.0);",
                "    } else {",
                "        gl_FragColor = sceneColor;",
                "    }",
                "}"
            ].join("\n")
        });

        this.resolution = new Float32Array(2);

        this.outlineThickness = 1.0;
        this.outlineColor = [0, 0, 0];
        this.outlineDepthThreshold = 0.005;
        this.nearClip = 0.1;
        this.farClip = 1000;
        this.maxOutlineDistance = 100;
    }

    render(inputTarget, outputTarget, rect) {
        const device = this.device;
        const scope = device.scope;

        scope.resolve("uColorBuffer").setValue(inputTarget.colorBuffer);

        if (device.sceneDepthMap) {
            scope.resolve("uDepthBuffer").setValue(device.sceneDepthMap);
        } else {
            scope.resolve("uDepthBuffer").setValue(inputTarget.colorBuffer);
        }

        this.resolution[0] = inputTarget.width;
        this.resolution[1] = inputTarget.height;
        scope.resolve("uResolution").setValue(this.resolution);

        scope.resolve("uOutlineThickness").setValue(this.outlineThickness);
        scope.resolve("uOutlineColor").setValue(this.outlineColor);
        scope.resolve("uOutlineDepthThreshold").setValue(this.outlineDepthThreshold);
        scope.resolve("uNearClip").setValue(this.nearClip);
        scope.resolve("uFarClip").setValue(this.farClip);
        scope.resolve("uMaxOutlineDistance").setValue(this.maxOutlineDistance);

        pc.drawFullscreenQuad(device, outputTarget, pc.PostEffect.quadVertexBuffer, this.shader, rect);
    }
}

/**
 * @class
 * @name outline
 * @classdesc
 * [EN] PlayCanvas script to add `OutlinePostEffect` to a camera.  
 * [JP] カメラに `OutlinePostEffect` を追加する PlayCanvas スクリプト。
 *
 * @description
 * [EN] Attach this script to a camera entity to enable outline rendering.  
 * Attributes allow customization of thickness, color, depth threshold, and distance.  
 *
 * [JP] このスクリプトをカメラエンティティにアタッチすると、アウトライン描画が有効になります。  
 * 属性で太さ・色・しきい値・距離などを調整できます。
 *
 * @property {number} outlineThickness - Outline Thickness / アウトラインの太さ  
 *    EN: Outline thickness in pixels.  
 *    JP: アウトラインの太さ（ピクセル単位）。
 *
 * @property {pc.Color} outlineColor - Outline Color / アウトラインの色  
 *    EN: Color of the outline.  
 *    JP: アウトラインの描画色。
 *
 * @property {number} outlineDepthThreshold - Depth Threshold / 深度しきい値  
 *    EN: Depth difference to trigger outline.  
 *    JP: 輪郭を検出するための深度差。
 * 　　0.05ぐらいがちょうど良い。
 *    値が小さいほどすぐ深度差を検知し、線だらけになる。
 *    値が大きくなると、深度差が検知しにくくなり、線が少なくなる。
 *
 * @property {number} maxOutlineDistance - Max Outline Distance / 最大表示距離  
 *    EN: Outline fades beyond this distance.  
 *    JP: この距離を超えるとアウトラインがフェードします。
 *    値が小さいほど、近くのものにしか線がつかない。
 *    値が大きくなると、ほぼ全てのものに線がつく。
 */
var PostEffectOutline = pc.createScript('outline');

PostEffectOutline.attributes.add('outlineThickness', {
    type: 'number',
    default: 1,
    min: 1,
    max: 4,
    title: 'Outline Thickness / アウトラインの太さ'
});

PostEffectOutline.attributes.add('outlineColor', {
    type: 'rgb',
    default: [0, 0, 0],
    title: 'Outline Color / アウトラインの色'
});

PostEffectOutline.attributes.add('outlineDepthThreshold', {
    type: 'number',
    default: 0.005,
    min: 0.0001,
    max: 0.5,
    step: 0.0001,
    title: 'Depth Threshold / 深度閾値'
});

PostEffectOutline.attributes.add('maxOutlineDistance', {
    type: 'number',
    default: 100,
    min: 1,
    max: 10000,
    title: 'Max Outline Distance / 最大表示距離'
});

PostEffectOutline.prototype.initialize = function() {
    const effect = new OutlinePostEffect(this.app.graphicsDevice);
    this.effect = effect;

    this.entity.camera.enableDepthTexture = true;

    const queue = this.entity.camera.postEffects;
    queue.addEffect(effect);

    // 初期値設定
    effect.outlineThickness = this.outlineThickness;
    effect.outlineColor = [this.outlineColor.r, this.outlineColor.g, this.outlineColor.b];
    effect.outlineDepthThreshold = this.outlineDepthThreshold;
    effect.nearClip = this.entity.camera.nearClip;
    effect.farClip = this.entity.camera.farClip;
    effect.maxOutlineDistance = this.maxOutlineDistance;

    // 属性変更イベント
    this.on('attr:outlineThickness', value => { effect.outlineThickness = value; });
    this.on('attr:outlineColor', value => { effect.outlineColor = [value.r, value.g, value.b]; });
    this.on('attr:outlineDepthThreshold', value => { effect.outlineDepthThreshold = value; });
    this.on('attr:maxOutlineDistance', value => { effect.maxOutlineDistance = value; });

    this.on('enable', () => { queue.addEffect(effect); });
    this.on('disable', () => { queue.removeEffect(effect); });
};
