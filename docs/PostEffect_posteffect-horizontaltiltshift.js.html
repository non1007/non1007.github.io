// --------------- POST EFFECT DEFINITION --------------- //
/**
 * @class
 * @name HorizontalTiltShiftEffect
 * @classdesc Simple fake tilt-shift effect, modulating two-pass Gaussian blur by horizontal position.
 * [EN] Applies a horizontal tilt-shift blur effect by modulating Gaussian blur strength based on horizontal distance from a focal point.  
 * [JP] 水平方向の位置に応じてガウスぼかしの強さを変化させることで、簡易的なチルトシフト効果を適用します。
 *
 * @augments PostEffect
 * @param {GraphicsDevice} graphicsDevice
 * [EN] The graphics device of the application.  
 * [JP] アプリケーションの GraphicsDevice。
 *
 * @property {number} focus
 * [EN] Horizontal focal point (0.0 = left edge, 1.0 = right edge).  
 * [JP] 水平方向の焦点位置（0.0 = 左端、1.0 = 右端）。
 */
class HorizontalTiltShiftEffect extends pc.PostEffect {
    constructor(graphicsDevice) {
        super(graphicsDevice);

        const fshader = `
            uniform sampler2D uColorBuffer;
            uniform float uH;
            uniform float uR;
            varying vec2 vUv0;

            void main() {
                vec4 sum = vec4(0.0);
                float hh = uH * abs(uR - vUv0.x);

                sum += texture2D(uColorBuffer, vec2(vUv0.x - 4.0 * hh, vUv0.y)) * 0.051;
                sum += texture2D(uColorBuffer, vec2(vUv0.x - 3.0 * hh, vUv0.y)) * 0.0918;
                sum += texture2D(uColorBuffer, vec2(vUv0.x - 2.0 * hh, vUv0.y)) * 0.12245;
                sum += texture2D(uColorBuffer, vec2(vUv0.x - 1.0 * hh, vUv0.y)) * 0.1531;
                sum += texture2D(uColorBuffer, vec2(vUv0.x, vUv0.y)) * 0.1633;
                sum += texture2D(uColorBuffer, vec2(vUv0.x + 1.0 * hh, vUv0.y)) * 0.1531;
                sum += texture2D(uColorBuffer, vec2(vUv0.x + 2.0 * hh, vUv0.y)) * 0.12245;
                sum += texture2D(uColorBuffer, vec2(vUv0.x + 3.0 * hh, vUv0.y)) * 0.0918;
                sum += texture2D(uColorBuffer, vec2(vUv0.x + 4.0 * hh, vUv0.y)) * 0.051;

                gl_FragColor = sum;
            }
        `;

        this.shader = pc.ShaderUtils.createShader(graphicsDevice, {
            uniqueName: 'HorizontalTiltShiftShader',
            attributes: { aPosition: pc.SEMANTIC_POSITION },
            vertexGLSL: pc.PostEffect.quadVertexShader,
            fragmentGLSL: fshader
        });

        // uniforms
        this.focus = 0.35;
    }

    render(inputTarget, outputTarget, rect) {
        const scope = this.device.scope;
        scope.resolve('uH').setValue(1 / inputTarget.width);
        scope.resolve('uR').setValue(this.focus);
        scope.resolve('uColorBuffer').setValue(inputTarget.colorBuffer);
        this.drawQuad(outputTarget, this.shader, rect);
    }
}

// ----------------- SCRIPT DEFINITION ------------------ //
/**
 * @class
 * @name HorizontalTiltShift
 * @classdesc PlayCanvas script wrapper for {@link HorizontalTiltShiftEffect}.
 * [EN] Script component to use HorizontalTiltShiftEffect in the PlayCanvas Editor.  
 * [JP] PlayCanvas エディタで HorizontalTiltShiftEffect を利用するためのスクリプトコンポーネント。
 *
 * @property {number} focus
 * [EN] Horizontal focal point for the blur (0.0 = left edge, 1.0 = right edge).  
 * [JP] ぼかしの基準となる水平方向の焦点位置（0.0 = 左端、1.0 = 右端）。
 */
var HorizontalTiltShift = pc.createScript('horizontalTiltShift');

HorizontalTiltShift.attributes.add('focus', {
    type: 'number',
    default: 0.35,
    min: 0,
    max: 1,
    title: 'Focus / フォーカス位置',
    description: 'EN: Horizontal focal point (0.0 = left, 1.0 = right). / JP: 水平方向の焦点位置（0.0 = 左端, 1.0 = 右端）。'
});

HorizontalTiltShift.prototype.initialize = function () {
    this.effect = new HorizontalTiltShiftEffect(this.app.graphicsDevice);
    this.effect.focus = this.focus;

    this.on('attr:focus', function (value) {
        this.effect.focus = value;
    }, this);

    const queue = this.entity.camera.postEffects;
    queue.addEffect(this.effect);

    this.on('state', function (enabled) {
        if (enabled) {
            queue.addEffect(this.effect);
        } else {
            queue.removeEffect(this.effect);
        }
    });

    this.on('destroy', function () {
        queue.removeEffect(this.effect);
    });
};
